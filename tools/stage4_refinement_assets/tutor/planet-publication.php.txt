<?php
/** Publikasi hanya dari daftar misi. Validasi hak akses, CSRF, soal, dan transaksi server. */
require __DIR__ . '/_auth.php';
require_once __DIR__ . '/../config/question_repository.php';
$tutor = tutor_require_login();
if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    http_response_code(405);header('Allow: POST');exit('Gunakan metode POST.');
}
if (!tutor_valid_csrf($_POST['csrf_token'] ?? null)) {
    http_response_code(403);exit('Token keamanan tidak valid. Muat ulang daftar misi.');
}
$id = isset($_POST['planet_id']) && is_string($_POST['planet_id']) ? trim($_POST['planet_id']) : '';
if (strlen($id) > 64 || !preg_match('/^[a-z0-9]+(?:-[a-z0-9]+)*$/D', $id)) {
    http_response_code(400);exit('ID planet tidak valid.');
}
$action = $_POST['action'] ?? null;
if (!in_array($action, ['publish','draft'], true)) {
    http_response_code(400);exit('Aksi tidak dikenal.');
}
$pdo = null;
$flash = ['ok'=>false,'message'=>'Publikasi gagal.'];
try {
    $pdo = mission_db();
    $pdo->beginTransaction();
    $stmt = $pdo->prepare('SELECT id,created_by FROM bs_planets WHERE id=? FOR UPDATE');
    $stmt->execute([$id]);
    $planet = $stmt->fetch(PDO::FETCH_ASSOC);
    if (!$planet || ($tutor['role']!=='admin' && (int)$planet['created_by']!==(int)$tutor['id'])) {
        $pdo->rollBack();http_response_code(404);exit('Planet tidak tersedia untuk tutor ini.');
    }
    if ($action === 'publish') {
        // Hitung hanya soal dengan mode yang sama dengan misi; soal lama dari mode lain diabaikan.
        $stmt = $pdo->prepare('SELECT m.id, m.mode_type, COUNT(q.id) AS questions
            FROM bs_missions m
            LEFT JOIN bs_mission_questions q ON q.mission_id=m.id AND q.mode_type=m.mode_type
            WHERE m.planet_id=? GROUP BY m.id, m.mode_type');
        $stmt->execute([$id]);
        $rows = $stmt->fetchAll(PDO::FETCH_ASSOC);
        $totalQuestions = array_sum(array_map(static fn($r)=>(int)$r['questions'],$rows));
        if ($totalQuestions < 1) {
            $pdo->rollBack();
            $flash=['ok'=>false,'message'=>'Planet tetap Draft. Buat minimal satu soal yang sesuai dengan gamemode misi.'];
        } else {
            $pdo->prepare("UPDATE bs_planets SET publication_status='published' WHERE id=?")->execute([$id]);
            $pdo->prepare("UPDATE bs_missions SET publication_status='published' WHERE planet_id=?")->execute([$id]);
            $pdo->commit();
            $flash=['ok'=>true,'message'=>'Planet berhasil dipublikasikan! Siswa yang memenuhi level dapat melihatnya. Misi siap dimainkan jika jumlah soal mencapai target.'];
        }
    } else {
        $pdo->prepare("UPDATE bs_planets SET publication_status='draft' WHERE id=?")->execute([$id]);
        $pdo->prepare("UPDATE bs_missions SET publication_status='draft' WHERE planet_id=?")->execute([$id]);
        $pdo->commit();
        $flash=['ok'=>true,'message'=>'Planet kembali menjadi Draft dan tersembunyi dari siswa.'];
    }
} catch (Throwable $e) {
    if ($pdo && $pdo->inTransaction()) $pdo->rollBack();
    error_log('BRIGHT SPACE publish error: '.$e->getMessage());
    $flash=['ok'=>false,'message'=>'Tidak dapat mengubah status. Periksa database dan log PHP.'];
}
$_SESSION['bs_publication_flash']=$flash;
header('Location: mission-list.php?planet='.rawurlencode($id), true, 303);
exit;
